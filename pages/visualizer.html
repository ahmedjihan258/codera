Loop Syntax Reference
/* for loop */
for (let i = start; i < end; i += step) {
  console.log("i =", i);
}

/* while loop */
let i = start;
while (i < end) {
  console.log("i =", i);
  i += step;
}

/* for...of loop (arrays) */
const arr = [10, 20, 30];
for (const val of arr) {
  console.log(val);
}
